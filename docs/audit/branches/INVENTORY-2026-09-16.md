## پیوست — بازتولید

هسته‌ی روش، برای هرکس خواست عددها را دوباره بسنجد. **بندِ پین مهم‌ترین قسمت است**؛ بدونش، pushِ یک همتا وسطِ کار ستون‌های شما را از دو لحظه‌ی متفاوت پر می‌کند:

```bash
git fetch --prune origin
# پین: هر origin/X را در refs/snap/X کپی کن تا fetchِ بعدیِ همتاها تکانش ندهد
git for-each-ref --format="%(objectname) %(refname:short)" refs/remotes/origin |
  grep -v -E " origin$| origin/HEAD$" |
  while read sha name; do git update-ref "refs/snap/${name#origin/}" "$sha"; done
MAIN=$(git rev-parse refs/snap/main)

# در یک worktreeِ دورریختنی، نه در چک‌اوتِ کسی
git worktree add --detach /tmp/snap "$MAIN"
for ref in $(git for-each-ref --format="%(refname:short)" refs/snap | grep -v "^snap/main$"); do
  b="refs/$ref"                             # refs/snap/<نامِ شاخه>
  git rev-list --count "$MAIN..$b"          # جلو
  git rev-list --count "$b..$MAIN"          # عقب
  git cherry "$MAIN" "$b"                   # «-» یعنی patchش بالادست هست
  git diff --name-only "$MAIN...$b"         # چه فایل‌هایی دست زده
  git diff --name-only "$MAIN" "$b"         # کدامشان هنوز فرق دارند
  git -C /tmp/snap merge --no-commit --no-ff "$b"
  git -C /tmp/snap diff --name-only --diff-filter=U | wc -l
  git -C /tmp/snap merge --abort; git -C /tmp/snap reset --hard "$MAIN"; git -C /tmp/snap clean -fd
done

# کارِ push‌نشده که هیچ جدولِ origin‌محوری نمی‌بیندش
git for-each-ref --format="%(refname:short)" refs/heads |
  while read n; do echo "$n $(git rev-list --count "$n" --not --remotes=origin)"; done
```
